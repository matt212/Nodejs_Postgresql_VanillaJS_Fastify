    entry_customer --> iam_audit
    iam_audit -- "On Validation Success" --> session_token
    session_token --> geo_request
    geo_request --> zone_map
    zone_map --> entry_staging
    entry_staging --> stock_check

    cart_state --> slot_req
    slot_req --> slot_alloc
    slot_alloc --> timestamps
    timestamps --> entry_order
    entry_order --> payment_proc
    payment_proc -- "On Payment Failure" --> rollback_error["Step 5.5: System - 'Transaction Rollback' & Stock Release"]

    payment_proc -- "On Payment Success" --> fund_auth
    fund_auth --> order_id
    order_id --> log_opt
    log_opt --> courier_id
    courier_id --> entry_manifest
    entry_manifest --> stock_reserve
    stock_reserve --> physical_pick
    physical_pick --> final_stock
    final_stock --> entry_handover
    entry_handover --> handover_check
    handover_check -- "On Item Mismatch" --> post_actions
    handover_check -- "On Success" --> final_cap
    final_cap --> delivered_status
    delivered_status --> post_actions
    post_actions --> end_state["Step 9.0: System - 'Closed Loop' Notification"]
