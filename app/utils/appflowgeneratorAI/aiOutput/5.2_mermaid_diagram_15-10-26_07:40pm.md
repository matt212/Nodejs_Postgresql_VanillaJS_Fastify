    subgraph phase_loc["Phase 02: Geolocation & Zone Validation"]

    item_verify --> stock_update

    exc_refund -.-> sys_terminal

    pay_gateway -- "Hard Failure: Declined" --> exc_dispute
    dispatcher_check -- "Hard Failure: No Couriers" --> exc_cancel
    loc_service -- "Hard Failure: Bad Address" --> exc_cancel
    stock_check -- "Hard Failure: Out of Stock" --> exc_cancel

        slot_degrade["Step 6.6: Degraded Path - Extended Window"]

    slot_assigned --> slot_reassign
    slot_reassign -- "Re-assignment Success" --> heal_phase06_exit
    slot_reassign -- "Re-assignment Fail" --> slot_cache
    slot_cache -- "Use Default" --> heal_phase06_exit
    slot_cache -- "Default Unavailable" --> slot_degrade
    slot_degrade -- "Accept Extended" --> heal_phase06_exit
    slot_degrade -- "User Rejects" --> exc_cancel
